# Builds the NoiseGenerator shutdown stress test twice: against the stock Nav2 source
# (STOCK_DIR, as in git HEAD) and against the same source with the shutdown fix (FIXED_DIR).
cmake_minimum_required(VERSION 3.16)
project(hang_check CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(rclcpp REQUIRED)
find_package(rclcpp_lifecycle REQUIRED)
find_package(nav2_ros_common REQUIRED)
find_package(Eigen3 REQUIRED)
set(MPPI_SRC /ws/src/navigation2/nav2_mppi_controller)
foreach(v stock fixed)
  foreach(san none tsan)
    set(t stress_${v}_${san})
    add_executable(${t} stress.cpp /tmp/hc/${v}/noise_generator.cpp ${MPPI_SRC}/src/parameters_handler.cpp)
    target_include_directories(${t} BEFORE PRIVATE /tmp/hc/stock/include)
    target_include_directories(${t} PRIVATE ${MPPI_SRC}/include)
    target_link_libraries(${t} rclcpp::rclcpp rclcpp_lifecycle::rclcpp_lifecycle
      nav2_ros_common::nav2_ros_common Eigen3::Eigen)
    target_compile_options(${t} PRIVATE -O2 -g)
    if(san STREQUAL "tsan")
      target_compile_options(${t} PRIVATE -fsanitize=thread)
      target_link_options(${t} PRIVATE -fsanitize=thread)
    endif()
  endforeach()
endforeach()
