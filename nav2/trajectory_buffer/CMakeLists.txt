cmake_minimum_required(VERSION 3.16)
project(trajectory_buffer)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
add_executable(trajectory_player src/trajectory_player.cpp)
target_link_libraries(trajectory_player rclcpp::rclcpp ${nav_msgs_TARGETS} ${geometry_msgs_TARGETS})
install(TARGETS trajectory_player DESTINATION lib/${PROJECT_NAME})
ament_package()
